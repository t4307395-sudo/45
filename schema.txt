-- سوبر ويب — سكيمة قاعدة بيانات Cloudflare D1
-- شغّل الملف ده مرة واحدة على قاعدة الـ D1 المربوطة بالمشروع (باسم binding: DB)
-- عن طريق: wrangler d1 execute <DB_NAME> --remote --file=schema.sql
-- أو من تبويب D1 في لوحة Cloudflare مباشرة (Console -> Run SQL)
-- ============================================================

-- المستخدمين (تسجيل بجوجل أو بإيميل/باسورد)
CREATE TABLE IF NOT EXISTS users (
    id             INTEGER PRIMARY KEY AUTOINCREMENT,
    email          TEXT UNIQUE NOT NULL,
    name           TEXT,
    password_hash  TEXT,
    google_id      TEXT UNIQUE,
    auth_provider  TEXT NOT NULL DEFAULT 'email', -- 'email' أو 'google'
    picture        TEXT,
    created_at     INTEGER DEFAULT (unixepoch() * 1000)
);

-- الحد اليومي لعدد الفحوصات + فترة الانتظار بين كل فحص والتاني
CREATE TABLE IF NOT EXISTS scan_limits (
    identifier     TEXT NOT NULL,           -- الإيميل حالياً
    scan_date      TEXT NOT NULL,           -- YYYY-MM-DD
    count          INTEGER NOT NULL DEFAULT 0,
    last_scan_at   INTEGER,                 -- Unix ms، لحساب فترة الانتظار (Cooldown)
    PRIMARY KEY (identifier, scan_date)
);

-- سجل تاريخ الفحوصات (لمقارنة "قبل وبعد")
CREATE TABLE IF NOT EXISTS scan_history (
    id                 INTEGER PRIMARY KEY AUTOINCREMENT,
    identifier         TEXT NOT NULL,
    url                TEXT NOT NULL,
    performance_score  INTEGER,
    seo_score          INTEGER,
    checked_at         INTEGER NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_scan_history_lookup ON scan_history (identifier, url, checked_at DESC);

-- كاش نتائج الفحص (ساعة واحدة لكل رابط)
CREATE TABLE IF NOT EXISTS analysis_cache (
    url          TEXT PRIMARY KEY,
    result_json  TEXT NOT NULL,
    cached_at    INTEGER NOT NULL
);

-- رموز ربط Google Drive/Sheets الخاصة بكل مستخدم
CREATE TABLE IF NOT EXISTS google_tokens (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    user_email      TEXT UNIQUE NOT NULL,
    access_token    TEXT NOT NULL,
    refresh_token   TEXT,
    expires_at      INTEGER NOT NULL,
    spreadsheet_id  TEXT
);

-- عداد تدوير مفاتيح Gemini (صف واحد بس)
CREATE TABLE IF NOT EXISTS api_key_rotation (
    id       INTEGER PRIMARY KEY,
    counter  INTEGER NOT NULL DEFAULT 0
);
INSERT OR IGNORE INTO api_key_rotation (id, counter) VALUES (1, 0);
